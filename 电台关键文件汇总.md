// "XX명이 함께 듣는 중" 社交条用的假听众
export const RADIO_LISTENERS: Listener[] = [
  { name: '토끼', emoji: '🐰', reaction: '💛' },
  { name: '늑대', emoji: '🐺', reaction: '🔥' },
  { name: '곰', emoji: '🐻' },
  { name: '여우', emoji: '🦊', reaction: '👏' },
];

// 一起听人数（假数据）
export const RADIO_LISTENER_COUNT = 23;

// 节目固定播出顺序（按时段），编成表/上一首下一首都按这个排
export const PROGRAM_ORDER = [
  'squirrel-morning',
  'animal-news',
  'bear-night',
  'fox-cafe',
] as const;

// 已囤内容的最大 Day（内容尚未覆盖的 Day 回落到这个天数）
export function getMaxRadioDay(): number {
  return RADIO_EPISODES.reduce((m, e) => Math.max(m, e.day), 1);
}

// 把用户当前 Day 夹到已有内容范围内（Day 1..maxDay）
export function clampRadioDay(day: number): number {
  const max = getMaxRadioDay();
  if (!Number.isFinite(day) || day < 1) return 1;
  return Math.min(day, max);
}

// 某个 Day 的 4 档节目，按播出时段排序
export function getEpisodesForDay(day: number): RadioEpisode[] {
  const d = clampRadioDay(day);
  return RADIO_EPISODES.filter((e) => e.day === d).sort(
    (a, b) => PROGRAM_ORDER.indexOf(a.program as never) - PROGRAM_ORDER.indexOf(b.program as never),
  );
}

export function getRadioEpisode(id: string): RadioEpisode | undefined {
  return RADIO_EPISODES.find((e) => e.id === id);
}

// 首页 "正在播放"：按当前钟点挑当天最贴近的一档
export function getNowPlaying(day: number): RadioEpisode {
  const list = getEpisodesForDay(day);
  const nowHm = new Date().getHours() * 60 + new Date().getMinutes();
  let pick = list[0];
  for (const ep of list) {
    const [h, m] = ep.scheduleTime.split(':').map(Number);
    if (h * 60 + m <= nowHm) pick = ep;
  }
  return pick;
}

// scheduleTime("HH:MM") 换算成当天分钟数
export function scheduleMinutes(ep: RadioEpisode): number {
  const [h, m] = ep.scheduleTime.split(':').map(Number);
  return h * 60 + m;
}

// 实时编成模式：某档是否已到播出时间（nowHm = 当前分钟数）
export function isReleased(ep: RadioEpisode, nowHm: number): boolean {
  return scheduleMinutes(ep) <= nowHm;
}

// 播出时间的韩语公开标签，如 "오전 7시 공개" / "오후 12시 공개"
export function releaseLabel(ep: RadioEpisode): string {
  const [h, m] = ep.scheduleTime.split(':').map(Number);
  const period = h < 12 ? '오전' : '오후';
  const h12 = h <= 12 ? h : h - 12;
  const mm = m > 0 ? ` ${m}분` : '';
  return `${period} ${h12}시${mm} 공개`;
}

import { NextResponse } from 'next/server';
